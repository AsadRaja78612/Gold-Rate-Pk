async function getRate(){ let res = await fetch("https://metals-api.com/api/latest"); let data = await res.json(); // ✔️ REAL GOLD PRICE (IMPORTANT FIX) let usdPerOunce = data.rates["Gold(USDXAU)"]; let usdToPkr = 280; const OUNCE_TO_GRAM = 31.1035; const GRAM_TO_TOLA = 11.6638; // conversions let usdPerGram = usdPerOunce / OUNCE_TO_GRAM; let pkrPerGram = usdPerGram * usdToPkr; let pkrPerTola = pkrPerGram * GRAM_TO_TOLA; // UI document.getElementById("usdOunce").innerText = usdPerOunce.toFixed(2); document.getElementById("gram").innerText = pkrPerGram.toFixed(0); document.getElementById("tola").innerText = pkrPerTola.toFixed(0); }
